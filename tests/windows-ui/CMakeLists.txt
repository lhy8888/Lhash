cmake_minimum_required(VERSION 3.20)
project(LHashWindowsCrashDriver LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic driver requires native 64-bit Windows; build the x64 configuration.")
endif()
add_executable(lhash_windows_crash_driver LHashCrashDriver.cpp)
target_compile_features(lhash_windows_crash_driver PRIVATE cxx_std_17)
target_compile_definitions(lhash_windows_crash_driver PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(lhash_windows_crash_driver PRIVATE Dbghelp User32)
if(MSVC)
    target_compile_options(lhash_windows_crash_driver PRIVATE /W4 /utf-8)
endif()
