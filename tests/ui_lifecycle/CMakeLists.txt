cmake_minimum_required(VERSION 3.20)
project(LHashUiLifecycleTests LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/LHashAppleSilicon.cmake")
find_package(Threads REQUIRED)
enable_testing()

add_executable(lhash_ui_lifecycle_tests
    lifecycle_tests.cpp
    ../../trunk/source/Common/strhelper.cpp)
target_compile_features(lhash_ui_lifecycle_tests PRIVATE cxx_std_17)
target_include_directories(lhash_ui_lifecycle_tests PRIVATE include ../../trunk/source)
target_link_libraries(lhash_ui_lifecycle_tests PRIVATE Threads::Threads)
if(APPLE)
    target_link_libraries(lhash_ui_lifecycle_tests PRIVATE iconv)
endif()
if(MSVC)
    target_compile_options(lhash_ui_lifecycle_tests PRIVATE /utf-8)
    target_compile_definitions(lhash_ui_lifecycle_tests PRIVATE UNICODE _UNICODE)
endif()
add_test(NAME lhash_ui_lifecycle_tests COMMAND lhash_ui_lifecycle_tests)
set_tests_properties(lhash_ui_lifecycle_tests PROPERTIES TIMEOUT 30)

if(WIN32)
    add_executable(lhash_worker_handle_tests worker_handle_tests.cpp ../../trunk/source/Common/strhelper.cpp)
    target_compile_features(lhash_worker_handle_tests PRIVATE cxx_std_17)
    target_include_directories(lhash_worker_handle_tests PRIVATE include ../../trunk/source)
    target_compile_definitions(lhash_worker_handle_tests PRIVATE UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(lhash_worker_handle_tests PRIVATE /utf-8)
    endif()
    add_test(NAME lhash_worker_handle_tests COMMAND lhash_worker_handle_tests)
    set_tests_properties(lhash_worker_handle_tests PROPERTIES TIMEOUT 30)
endif()
